==================
labels, break and continue
==================

class C {
  void f() {
    int i = 0
    outer:
    for ( : : ) {
      i++
      for (int j = 0 : j < 3 : j++) {
        if (j == 1) {
          continue outer
        }
        if (i > 2) {
          break outer
        }
      }
    }
  }
}

      System.out.println("body")

      System.out.println("done")
    }
    try (make("expr")
         local) {
      System.out.println("in body")
    } catch (Exception e) {
      System.out.println(e)
    }
  }
}
